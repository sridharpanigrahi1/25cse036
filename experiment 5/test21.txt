interface Draw {
    void draw();
}

class Circle implements Draw {
    public void draw() {
        System.out.println("Drawing a Circle");
    }
}

class Rectangle implements Draw {
    public void draw() {
        System.out.println("Drawing a Rectangle");
    }
}

public class test21 {
    public static void main(String[] args) {
        Draw d;

        d = new Circle();
        d.draw();

        d = new Rectangle();
        d.draw();
    }
}
